# Player convenience: muster one Supreme Warriors regiment without Military UI grind.
# Cavalry (supreme_riders) and optional fourth archetype remain Military UI only.

supreme_muster_warriors_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major
	desc = supreme_muster_warriors_decision_desc
	selection_tooltip = supreme_muster_warriors_decision_tooltip
	confirm_text = supreme_muster_warriors_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_alive = yes
		is_ai = no
		is_ruler = yes
	}

	is_valid = {
		number_maa_regiments_of_type = {
			target = supreme_warriors
			value < 3
		}
	}

	cost = {
		gold = 50
	}

	effect = {
		create_maa_regiment = {
			type = supreme_warriors
			check_can_recruit = no
			size = 1
		}
		sw_sync_warriors_banner_effect = yes
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}

# Permanent army mobility doctrine (land + sea/embark). One-shot via modifier gate.

sw_warrior_mobility_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	decision_group_type = major
	desc = sw_warrior_mobility_decision_desc
	selection_tooltip = sw_warrior_mobility_decision_tooltip
	confirm_text = sw_warrior_mobility_decision_confirm

	ai_check_interval = 0

	is_shown = {
		is_alive = yes
		is_ai = no
		is_ruler = yes
		NOT = { has_character_modifier = sw_warrior_mobility_modifier }
		sw_owns_any_elite_maa_trigger = yes
	}

	is_valid = {
		NOT = { has_character_modifier = sw_warrior_mobility_modifier }
		sw_owns_any_elite_maa_trigger = yes
	}

	effect = {
		add_character_modifier = {
			modifier = sw_warrior_mobility_modifier
			days = -1
		}
		sw_sync_warriors_banner_effect = yes
	}

	ai_potential = {
		always = no
	}

	ai_will_do = {
		base = 0
	}
}
